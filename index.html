문제 1: 두 정수 사이의 합 구하기

두 개의 정수를 매개변수로 입력받아 두 정수 사이에 있는 모든 정수의 합을 구하여 반환하는 함수를 정의하고, main함수에서 이를 호출하는 프로그램을 작성하세요.

실행결과 예시:

5와 9 사이의 정수의 총합은 : 21

#include <stdio.h>

int totalBetween(int a, int b) { //사용자 함수 선언 및 정의
    int total = 0; //합계
    for (int i = a + 1; i < b; i++) {
        total += i; //total = total + i;
    }
    return total;
}

int main(void) {
    int result = totalBetween(5, 9); //함수 호출
    printf("5와 9 사이의 정수의 총합은 : %d\n", result);
    return 0;
}

문제 2: 7의 배수 개수 구하기

한 개의 정수를 매개변수로 입력받아 입력받은 수와 그 수보다 200 더 큰 정수 사이에 7의 배수가 몇 개나 존재하는지를 구하여 그 결과를 반환하는 함수를 구현하세요.

예를 들어 100을 전달받으면 100부터 300 사이의 7의 배수가 몇 개 존재하는지 구하면 됩니다.

실행결과 예시:

정수를 입력하세요 : 100

100과 300 사이의 7의 배수는 28개입니다!

#include <stdio.h>

int howManySeven(int number) {
    int howmany = 0; //7의 배수의 개수
    for (int i = number; i < number + 200; i++) {
        if (i % 7 == 0)
            howmany++; //1 증가
    }
    return howmany;
}

int main(void) {
    int number, result;
    printf("정수를 입력하세요 : ");
    scanf_s("%d", &number);
    result = howManySeven(number); //사용자 함수 호출
    printf("%d과 %d 사이의 7의 배수는 %d개!\n", number, number + 200, result);
    return 0;
}

문제 4: 팩토리얼 계산 함수

입력 예시: 5

출력 예시: 120

#include <stdio.h>

int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

    return 0;

문제 6: Call-by-value와 Call-by-reference

int형 변수 number1과 number2를 선언하고, 두 개의 인자를 전달받는 두 개의 함수를 정의해 보세요.

두 함수의 이름과 기능은 다음과 같습니다:

getDiffByValue: Call-by-value 방식으로 두 수의 차를 구하여 반환하는 함수

callDiffByRef: Call-by-reference 방식으로 두 수 중에서 큰 것을 number1에, 작은 것을 number2에 저장하는 함수 (number1에 저장된 정수가 더 큰 경우 변화가 없으나, number2에 저장된 정수가 더 큰 경우 교환)

HINT: Call-by-reference 함수의 매개 변수를 선언할 때는 포인터 연산자를 사용합니다.

#include <stdio.h>

int getDiffByValue(int num1, int num2) {
    return (num1 > num2) ? (num1 - num2) : (num2 - num1);
}

void callDiffByRef(int *num1, int *num2) {
    if (*num2 > *num1) {
        int temp = *num1;
        *num1 = *num2;
        *num2 = temp;
    }
}

int main(void) {
    int number1 = 10, number2 = 20;
    printf("차이: %d\n", getDiffByValue(number1, number2));
    callDiffByRef(&number1, &number2);
    printf("number1: %d, number2: %d\n", number1, number2);
    return 0;
}

문제 7: 가위바위보 게임

가위바위보 게임을 만들어 보겠습니다. 다음과 같이 규칙을 정하고 난수 발생 함수 srand를 사용해 1~3 중 하나의 숫자를 발생시킵니다.

가위: 1, 바위: 2, 보: 3

난수가 발생한 다음에는 사용자가 자신의 숫자를(1~3) 직접 입력하여 난수와 가위바위보를 한 결과를 출력해 보겠습니다.

가위바위보 게임은 함수로 만들고, main 함수에는 이에 대한 호출문 외에는 존재하지 않도록 프로그램을 작성해 보세요.

실행 예:

가위 1 바위 2 보 3 당신의 선택은 : 2

당신은 바위, 컴퓨터 난수는 바위, 비겼습니다.

풀이 1: if-else 문 사용

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void playGame() {
    const char* choices[] = {"", "가위", "바위", "보"};
    int user, computer;
    
    srand(time(NULL));
    computer = rand() % 3 + 1;
    
    printf("가위 1 바위 2 보 3 당신의 선택은 : ");
    scanf_s("%d", &user);
    
    printf("당신은 %s, 컴퓨터 난수는 %s, ", choices[user], choices[computer]);
    
    if (user == computer) {
        printf("비겼습니다.\n");
    } else if ((user == 1 && computer == 3) || 
               (user == 2 && computer == 1) || 
               (user == 3 && computer == 2)) {
        printf("이겼습니다!\n");
    } else {
        printf("졌습니다.\n");
    }
}

int main(void) {
    playGame();
    return 0;
}

문제 1: 포인터를 이용한 값 교환

두 변수의 값을 교환하는 함수 swap을 포인터를 이용하여 작성하세요.

입력 예시: a=10, b=20

출력 예시: a=20, b=10

#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 10, y = 20;
    printf("교환 전: x=%d, y=%d\n", x, y);
    swap(&x, &y);
    printf("교환 후: x=%d, y=%d\n", x, y);
    return 0;
}

문제 3: 이중 포인터 실행 결과 예측

다음 프로그램의 실행 결과를 예측해 보세요.

#include <stdio.h>

int main(void)
{
    int a = 33, b = 66;
    int *pa = &a;
    int *pb = &b;
    int **ppa = &pa;
    int **ppb = &pb;
    int *ptr;

    ptr = *ppb;
    *ppb = *ppa;
    *ppa = ptr;

    printf("%d %d %d %d\n", a, b, *pa, *pb);

    return 0;
}

실행 결과:

33 66 66 33

해설:

이 프로그램은 이중 포인터를 사용하여 포인터 변수들이 가리키는 대상을 교환합니다.

  1. 초기 상태:
    • a = 33, b = 66
    • pa는 a를 가리킴
    • pb는 b를 가리킴
    • ppa는 pa를 가리킴
    • ppb는 pb를 가리킴
  2. ptr = *ppb; → ptr에 pb의 값(b의 주소)을 저장
  3. *ppb = *ppa; → pb가 이제 a를 가리킴
  4. *ppa = ptr; → pa가 이제 b를 가리킴
  5. 결과적으로 pa와 pb가 가리키는 대상이 서로 바뀜
  6. 따라서 a는 33, b는 66, *pa는 66, *pb는 33

문제 4: 함수 포인터 선언

다음과 같이 선언된 함수들에 대해서 대입 연산이 가능하도록 빈칸에 함수 포인터를 선언해 보세요(함수의 정의부는 생략).

01      #include <stdio.h>
02
03      // 함수의 선언부
04      int FirstFunc(int a, int b);
05      double SecondFunc(double d);
06      void ThirdFunc(char * c);
07
08      int main(void)
09      {
10
11          // 이 부분에 코드를 작성하세요.
12          // (빈칸)
13          // (빈칸)
14
15          first = FirstFunc;
16          second = SecondFunc;
17          third = ThirdFunc;
18
19          return 0;
20      }

정답:

int (*first)(int, int);
double (*second)(double);
void (*third)(char *);

해설:

함수 포인터는 함수의 시그니처(반환형과 매개변수)와 일치하도록 선언해야 합니다.

    • 반환형: void
    • 매개변수: char * 1개
    • ThirdFunc와 시그니처가 일치

함수 포인터 선언 형식:

반환형 (*포인터이름)(매개변수타입1, 매개변수타입2, ...);

문제 2: 2차원 동적 배열

행과 열의 크기를 입력받아 2차원 동적 배열을 생성하고, 값을 입력받아 출력하는 프로그램을 작성하세요.

#include <stdio.h>
#include <stdlib.h>

int main() {
    int rows, cols;
    printf("행과 열 입력: ");
    scanf_s("%d %d", &rows, &cols);

    int** matrix = (int**)malloc(rows * sizeof(int*));
    for (int i = 0; i < rows; i++) {
        matrix[i] = (int*)malloc(cols * sizeof(int));
    }

    printf("값 입력:\n");
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            scanf_s("%d", &matrix[i][j]);
        }
    }

    printf("출력:\n");
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");

    char* s2 = (char*)malloc(strlen(s1) + 1);

    // s2: 목적지
    // strlen(s1) + 1: 목적지의 크기 (널 문자 포함)
    // s1: 복사할 원본
    strcpy_s(s2, strlen(s1) + 1, s1);

    printf("%s\n", s2);

    if (s2 != NULL) free(s2);

    return 0;
}

해설:

  • 빈칸 ①: char* s2 = (char*)malloc(strlen(s1) + 1);
    • strlen(s1): s1 문자열의 길이를 구함 (NULL 문자 제외)
    • + 1: NULL 문자('\0')를 저장할 공간 추가
    • malloc(): 필요한 크기만큼 동적 메모리 할당
    • (char*): 반환된 void 포인터를 char 포인터로 형변환
  • 빈칸 ②: strcpy_s(s2, strlen(s1) + 1, s1);
    • strcpy_s는 보안이 강화된 문자열 복사 함수입니다
    • 첫 번째 인자: 목적지 포인터 (s2)
    • 두 번째 인자: 목적지 버퍼의 크기 (strlen(s1) + 1)
    • 세 번째 인자: 복사할 원본 문자열 (s1)
    • 버퍼 오버플로우를 방지하여 더 안전합니다

strcpy vs strcpy_s:

  • strcpy: 표준 C 함수, 버퍼 크기 검사 없음
  • strcpy_s: 보안 강화 함수, 버퍼 크기를 명시하여 오버플로우 방지
  • Visual Studio에서는 strcpy_s 사용을 권장합니다

문제 5: 할당된 영역의 크기가 가장 큰 포인터 고르기

다음 코드에서 동적 메모리 할당이 완료된 후, 할당된 영역의 크기가 가장 큰 포인터를 고르세요.

#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    int arr[5] = {10, 20, 30, 40, 50};
    int *pa;
    int *pb;
    int *pc;
    int i;

    // 1. pa 할당
    pa = (int *)malloc(sizeof(arr));

    for(i = 0; i < 5; i++)
    {
        pa[i] = arr[i];
    }

    // 2. pb 할당
    pb = (int *)calloc(pa[3], sizeof(int));
    
    // 3. pc 할당
    pc = (int *)realloc(NULL, pa[4]);

    return 0;
}

질문: pa, pb, pc 중 할당된 메모리 크기가 가장 큰 것은?

정답: pb

해설:

각 포인터에 할당된 메모리 크기를 계산해 보겠습니다.

  • pa 할당: malloc(sizeof(arr))
    • sizeof(arr) = sizeof(int) × 5 = 4 × 5 = 20바이트
  • pb 할당: calloc(pa[3], sizeof(int))
    • pa[3] = 40 (배열에서 복사된 값)
    • calloc(40, sizeof(int)) = 40 × 4 = 160바이트
    • calloc은 모든 바이트를 0으로 초기화합니다
  • pc 할당: realloc(NULL, pa[4])
    • pa[4] = 50 (배열에서 복사된 값)
    • realloc(NULL, 50) = 50바이트
    • realloc의 첫 번째 인자가 NULL이면 malloc과 동일하게 동작

크기 비교:

  • pa: 20바이트
  • pb: 160바이트 ✓ (가장 큼)
  • pc: 50바이트

주요 개념:

  • malloc(size): size 바이트만큼 메모리 할당 (초기화 안 함)
  • calloc(n, size): n × size 바이트만큼 메모리 할당 (0으로 초기화)
  • realloc(ptr, size): ptr이 NULL이면 malloc과 동일, 아니면 크기 재조정

문제 6: 동적 배열을 이용한 평균과 표준편차 계산

사용자로부터 학생 수 n을 입력받아, malloc()을 사용하여 double형 배열을 동적으로 할당하세요. n명의 학생 점수를 입력받아 저장한 후, 평균과 표준편차를 계산하여 출력하고 메모리를 해제하세요.

표준편차 공식: √(Σ(x - 평균)² / n)

입력 예시:

5
85.5 90.0 78.5 92.0 88.0

출력 예시:

평균: 86.80
표준편차: 4.67
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int main() {
    int n;
    double *scores;
    double sum = 0, mean, variance = 0, std_dev;
    
    printf("학생 수 입력: ");
    scanf("%d", &n);
    
    // 동적 메모리 할당
    scores = (double*)malloc(n * sizeof(double));
    if (scores == NULL) {
        printf("메모리 할당 실패\n");
        return 1;
    }
    
    // 점수 입력
    printf("점수 입력: ");
    for (int i = 0; i < n; i++) {
        scanf("%lf", &scores[i]);
        sum += scores[i];
    }
    
    // 평균 계산
    mean = sum / n;
    
    // 분산 계산
    for (int i = 0; i < n; i++) {
        variance += pow(scores[i] - mean, 2);
    }
    variance /= n;
    
    // 표준편차 계산
    std_dev = sqrt(variance);
    
    printf("평균: %.2lf\n", mean);
    printf("표준편차: %.2lf\n", std_dev);
    
    // 메모리 해제
    free(scores);
    
    return 0;
}

2 3

1 2 3

#include <stdio.h>
#include <stdlib.h>

    int row, col;
    int **matrix, **transpose;

    // 원본 행렬 동적 할당

    for (int i = 0; i < row; i++) {
        matrix[i] = (int*)malloc(col * sizeof(int));
    }

    }
    
    // 원본 행렬 입력
    printf("행렬 원소 입력:\n");
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            scanf("%d", &matrix[i][j]);
        }
    }
    
    // 전치 행렬 계산
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {
            transpose[j][i] = matrix[i][j];
        }
    }
    
    // 원본 행렬 출력
    printf("\n원본 행렬:\n");
    for (int i = 0; i < row; i++) {
        for (int j = 0; j < col; j++) {

    printf("\n전치 행렬:\n");

        for (int j = 0; j < row; j++) {
            printf("%d ", transpose[i][j]);
        }
        printf("\n");
    }
    
    // 메모리 해제
    for (int i = 0; i < row; i++) {
        free(matrix[i]);
    }
    free(matrix);
    
    for (int i = 0; i < col; i++) {

    }

문제 2: 구조체 포인터 활용

#include <stdio.h>
#include <math.h>

struct Point {
    double x;
    double y;
};

double distance(struct Point *p1, struct Point *p2) {
    double dx = p2->x - p1->x;
    double dy = p2->y - p1->y;
    return sqrt(dx*dx + dy*dy);
}

int main() {
    struct Point p1 = {0.0, 0.0};
    struct Point p2 = {3.0, 4.0};
    
    printf("두 점 사이의 거리: %.2f\n", distance(&p1, &p2));
    return 0;
}
